### 2.1 数据分类

| 数据类型 | 更新频率 | 数据量级 | 存储方式 |
|---------|---------|---------|---------|
| 股票池/元数据 | 日更 | KB级 | MySQL |

```python
# ifind_fetcher.py

from abc import ABC, abstractmethod
from typing import List, Dict, Optional
import pandas as pd
from datetime import date
from iFinDPy import *

class IFindFetcher:
    """iFind 数据采集器"""
    
    def __init__(self, username: str, password: str):
        self.username = username
        self.password = password
        self._logged_in = False
    
    def login(self) -> bool:
        """登录 iFind"""
        result = THS_iFinDLogin(self.username, self.password)
        self._logged_in = result in {0, -201}
        return self._logged_in
    
    def logout(self):
        """登出"""
        THS_iFinDLogout()
        self._logged_in = False


class StockPoolFetcher(IFindFetcher):
    """股票池数据采集"""
    
    def get_pool_members(self, block_code: str, date: str) -> pd.DataFrame:
        """
        获取板块成分股
        
        Args:
            block_code: 板块代码，如 '001005010' (全A股)
            date: 日期 YYYYMMDD
        
        Returns:
            DataFrame with columns: [code, name, exchange]
        """
        result = THS_DR(
            "p03291",
            f"date={date};blockname={block_code};iv_type=allcontract",
            "p03291_f001:Y,p03291_f002:Y,p03291_f003:Y,p03291_f004:Y"
        )
        if result.errorcode != 0:
            raise Exception(f"获取板块成分失败: {result.errmsg}")
        
        df = result.data
        df.columns = ['seq', 'code', 'name', 'exchange']
        return df[['code', 'name', 'exchange']]


class MarketDataFetcher(IFindFetcher):
    """行情数据采集"""
    
    def get_daily_quotes(
        self, 
        codes: List[str], 
        start_date: str, 
        end_date: str,
        indicators: List[str] = None
    ) -> pd.DataFrame:
        """
        获取日线行情
        
        Args:
            codes: 证券代码列表
            start_date: 开始日期 YYYY-MM-DD
            end_date: 结束日期 YYYY-MM-DD
            indicators: 指标列表，默认 OHLCV
        
        Returns:
            DataFrame with columns: [trade_date, code, open, high, low, close, volume, amount]
        """
        if indicators is None:
            indicators = ['open', 'high', 'low', 'close', 'volume', 'amount']
        
        code_str = ','.join(codes)
        indi_str = ';'.join(indicators)
        
        result = THS_HQ(
            code_str,
            indi_str,
            'period:D,pricetype:1',  # 日线，前复权
            start_date,
            end_date
        )
        
        if result.errorcode != 0:
            raise Exception(f"获取行情失败: {result.errmsg}")
        
        return result.data
    
    def get_realtime_quotes(self, codes: List[str]) -> pd.DataFrame:
        """获取实时行情"""
        code_str = ','.join(codes)
        result = THS_RQ(
            code_str,
            'latest;open;high;low;volume;amount;changeRatio'
        )
        if result.errorcode != 0:
            raise Exception(f"获取实时行情失败: {result.errmsg}")
        return result.data


class FactorDataFetcher(IFindFetcher):
    """因子数据采集"""
    
    # 预定义的因子指标映射
    FACTOR_MAPPING = {
        # 估值因子
        'pe_ttm': 'ths_pe_ttm_stock',
        'pb': 'ths_pb_stock',
        'ps': 'ths_ps_stock',
        'market_cap': 'ths_market_value_stock',
        'float_cap': 'ths_float_mv_stock',
        'dividend_yield': 'ths_dividend_yield_stock',
        
        # 财务因子
        'roe': 'ths_roe_stock',
        'roa': 'ths_roa_stock',
        'gross_margin': 'ths_gross_profit_margin_stock',
        'net_margin': 'ths_net_profit_margin_stock',
        'eps': 'ths_eps_stock',
        
        # 成长因子
        'revenue_yoy': 'ths_or_yoy_stock',
        'profit_yoy': 'ths_np_yoy_stock',
        
        # 其他
        'turnover': 'ths_turnover_ratio_stock',
    }
    
    def get_basic_factors(
        self, 
        codes: List[str], 
        factors: List[str], 
        date: str
    ) -> pd.DataFrame:
        """
        获取单日因子数据
        
        Args:
            codes: 证券代码列表
            factors: 因子名称列表 (使用简称)
            date: 日期 YYYY-MM-DD
        """
        code_str = ','.join(codes)
        
        # 转换因子名称为 iFind 指标代码
        indicators = [self.FACTOR_MAPPING.get(f, f) for f in factors]
        indi_str = ';'.join(indicators)
        
        result = THS_BD(code_str, indi_str, date)
        
        if result.errorcode != 0:
            raise Exception(f"获取因子数据失败: {result.errmsg}")
        
        return result.data
    
    def get_factor_series(
        self,
        codes: List[str],
        factor: str,
        start_date: str,
        end_date: str
    ) -> pd.DataFrame:
        """获取因子时间序列"""
        code_str = ','.join(codes)
        indicator = self.FACTOR_MAPPING.get(factor, factor)
        
        result = THS_DS(
            code_str,
            indicator,
            '',
            'Fill:Blank',
            start_date,
            end_date
        )
        
        if result.errorcode != 0:
            raise Exception(f"获取因子序列失败: {result.errmsg}")
        
        return result.data


class PITDataFetcher(IFindFetcher):
    """PIT (Point-in-Time) 时点数据采集"""
    
    def get_pit_data(
        self,
        codes: List[str],
        indicator: str,
        calc_date: str,
        report_date: str,
        data_type: int = 1  # 1=累计, 2=单季
    ) -> pd.DataFrame:
        """
        获取时点数据，避免未来函数
        
        Args:
            codes: 证券代码列表
            indicator: PIT指标，如 'ths_np_atoopc_pit_stock'
            calc_date: 计算日期 (策略运行日)
            report_date: 报告期 YYYYMMDD
            data_type: 数据类型
        """
        code_str = ','.join(codes)
        params = f'{calc_date},{report_date},{data_type}'
        
        result = THS_BD(code_str, indicator, params)
        
        if result.errorcode != 0:
            raise Exception(f"获取PIT数据失败: {result.errmsg}")
        
        return result.data
```

### 4.2 数据更新调度

```python
# scheduler.py

from datetime import datetime, timedelta
from typing import Callable
import schedule
import time

class DataUpdateScheduler:
    """数据更新调度器"""
    
    def __init__(self, fetcher_factory: Callable):
        self.fetcher_factory = fetcher_factory
        self.jobs = []
    
    def add_daily_job(self, job_func: Callable, run_time: str = "18:00"):
        """
        添加日更任务
        
        Args:
            job_func: 任务函数
            run_time: 执行时间 HH:MM
        """
        schedule.every().day.at(run_time).do(job_func)
        self.jobs.append(job_func.__name__)
    
    def add_realtime_job(self, job_func: Callable, interval_seconds: int = 3):
        """添加实时任务"""
        schedule.every(interval_seconds).seconds.do(job_func)
    
    def run(self):
        """运行调度器"""
        while True:
            schedule.run_pending()
            time.sleep(1)


# 更新任务示例
class DailyUpdateTask:
    """日更新任务"""
    
    def __init__(self, fetcher, storage):
        self.fetcher = fetcher
        self.storage = storage
    
    def update_stock_pool(self):
        """更新股票池"""
        today = datetime.now().strftime('%Y%m%d')
        
        pools = {
            'all_a': '001005010',
            'hs300': '001005261',
            'zz500': '001005262',
        }
        
        for pool_name, block_code in pools.items():
            df = self.fetcher.get_pool_members(block_code, today)
            self.storage.save_pool_members(pool_name, df, today)
    
    def update_daily_quotes(self):
        """更新日线行情"""
        today = datetime.now().strftime('%Y-%m-%d')
        codes = self.storage.get_all_codes()
        
        # 分批获取，避免超限
        batch_size = 100
        for i in range(0, len(codes), batch_size):
            batch_codes = codes[i:i+batch_size]
            df = self.fetcher.get_daily_quotes(batch_codes, today, today)
            self.storage.append_daily_quotes(df)
    
    def update_factors(self):
        """更新因子数据"""
        today = datetime.now().strftime('%Y-%m-%d')
        codes = self.storage.get_all_codes()
        
        factors = ['pe_ttm', 'pb', 'roe', 'roa', 'market_cap', 'turnover']
        
        batch_size = 500  # 基础数据可以更大批次
        for i in range(0, len(codes), batch_size):
            batch_codes = codes[i:i+batch_size]
            df = self.fetcher.get_basic_factors(batch_codes, factors, today)
            self.storage.save_factors(df, today)
```

## 八、快速开始

### 8.1 环境配置

```yaml
# config/config.yaml
ifind:
  username: "your_username"
  password: "your_password"

database:
  host: "localhost"
  port: 5432
  database: "quant"
  user: "postgres"
  password: "password"

redis:
  host: "localhost"
  port: 6379

storage:
  data_path: "./data"
  
scheduler:
  daily_update_time: "18:00"
  timezone: "Asia/Shanghai"
```

### 8.2 初始化脚本

```bash
# 1. 安装依赖
pip install -r requirements.txt

# 2. 初始化数据库
python scripts/init_db.py

# 3. 回填历史数据 (首次运行)
python scripts/backfill.py --start 2020-01-01 --end 2024-08-03

# 4. 启动日更服务
python -m src.scheduler.daily_update
```
